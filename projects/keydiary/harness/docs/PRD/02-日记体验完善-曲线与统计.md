CREATE TABLE IF NOT EXISTS app_usage (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    date        TEXT NOT NULL,           -- '2026-05-25'
    time_bucket TEXT NOT NULL,           -- '14:05'
    app_name    TEXT NOT NULL,           -- 'Visual Studio Code'
    key_count   INTEGER NOT NULL DEFAULT 0,  -- 该桶内在此app中的击键数
    
    UNIQUE(date, time_bucket, app_name)

## 3. KPM 时间曲线

### 3.1 数据查询

```sql
-- 查询某天的 KPM 曲线数据
-- 将 5 分钟桶的 count 汇总，换算为每分钟
SELECT 
    time_bucket,
    SUM(count) as bucket_total,
    CAST(SUM(count) AS REAL) / 5 as kpm  -- 5分钟桶内总键数 / 5 = 平均KPM
FROM key_events
WHERE date = ?
GROUP BY time_bucket
ORDER BY time_bucket;
```

### 3.2 Tauri Command

```rust
#[tauri::command]
fn get_kpm_timeline(db: State<Database>, date: String) -> Vec<KpmPoint> {
    // 返回 [{time: "09:00", kpm: 45}, {time: "09:05", kpm: 72}, ...]
    db.query_kpm_timeline(&date)
}
```

### 3.3 前端渲染

使用 D3.js 绘制面积图（area chart）：

- X 轴：00:00 → 23:55（每 5 分钟一个点）
- Y 轴：KPM 值
- 填充色：低于平均值为蓝色区域，高于平均值为绿色区域
- 标注线：个人日均 KPM 水平线（虚线）
- Hover 交互：显示具体时间点的 KPM 值

**视觉参考**：类似 Apple Watch 的心率曲线，平滑过渡，有呼吸感。

```svelte
<!-- components/KpmTimeline.svelte -->
<script lang="ts">
  import * as d3 from 'd3';
  import { invoke } from '@tauri-apps/api/core';

  export let date: string;
  
  let svgEl: SVGElement;
  let data: {time: string, kpm: number}[] = $state([]);
  
  $effect(() => {
    loadData(date);
  });
  
  async function loadData(d: string) {
    data = await invoke('get_kpm_timeline', { date: d });
    renderChart();
  }
  
  function renderChart() {
    const svg = d3.select(svgEl);
    svg.selectAll('*').remove();
    
    const width = 700, height = 200;
    const margin = { top: 20, right: 20, bottom: 30, left: 40 };
    
    const x = d3.scalePoint()
      .domain(data.map(d => d.time))
      .range([margin.left, width - margin.right]);
    
    const y = d3.scaleLinear()
      .domain([0, d3.max(data, d => d.kpm) * 1.1])
      .range([height - margin.bottom, margin.top]);
    
    const area = d3.area()
      .x(d => x(d.time))
      .y0(height - margin.bottom)
      .y1(d => y(d.kpm))
      .curve(d3.curveMonotoneX);  // 平滑曲线
    
    svg.append('path')
      .datum(data)
      .attr('d', area)
      .attr('fill', 'rgba(99, 179, 237, 0.3)')
      .attr('stroke', '#4299e1')
      .attr('stroke-width', 2);
    
    // ... X/Y轴、hover tooltip 等
  }
</script>

<svg bind:this={svgEl} viewBox="0 0 700 200" class="kpm-chart"></svg>
```

## 8. daily_summary 自动更新

每次 Aggregator flush 时，同步更新 `daily_summary` 表：

```rust
fn update_daily_summary(db: &Connection, date: &str) {
    db.execute(
        "INSERT OR REPLACE INTO daily_summary (date, total_keystrokes, active_minutes, peak_kpm, updated_at)
         VALUES (
             ?1,
             (SELECT COALESCE(SUM(count), 0) FROM key_events WHERE date = ?1),
             (SELECT COUNT(DISTINCT time_bucket) * 5 FROM key_events WHERE date = ?1),
             (SELECT COALESCE(MAX(bucket_kpm), 0) FROM (
                 SELECT CAST(SUM(count) AS REAL) / 5 as bucket_kpm 
                 FROM key_events WHERE date = ?1 GROUP BY time_bucket
             )),
             datetime('now', 'localtime')
         )",
        params![date],
    );
}
```
